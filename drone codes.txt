size=get_world_size()-1
			
def till_all_grass():
	while (get_pos_x() !=0):
		move(West)
	while (get_pos_y() !=0):
		move(South)
	for i in range(size/2):
		#print(i) #this is just for debugging
		while (get_pos_y() != i): #goes down
			if get_ground_type()==Grounds.Soil:
				till()
			move(South)
		while (get_pos_x() != size-i): #goes right
			if get_ground_type()==Grounds.Soil:
				till()
			move(East)
		while (get_pos_y() != size-i): #goes up
			if get_ground_type()==Grounds.Soil:
				till()
			move(North)
		while (get_pos_x() != i): #goes left
			if get_ground_type()==Grounds.Soil:
				till()
			move(West)
		if get_ground_type()==Grounds.Soil:
			till()

def mine_all_even():
	while (get_pos_x() !=0):
		move(West)
	while (get_pos_y() !=0):
		move(South)
	min_depth = -8
	for i in range(size/2):
		#print(i) #this is just for debugging
		while (get_pos_y() != i): #goes down
			if min_depth > get_pos_z():
				min_depth = get_pos_z()
			move(South)
		while (get_pos_x() != size-i): #goes right
			if min_depth > get_pos_z():
				min_depth = get_pos_z()
			move(East)
		while (get_pos_y() != size-i): #goes up
			if min_depth > get_pos_z():
				min_depth = get_pos_z()
			move(North)
		while (get_pos_x() != i): #goes left
			if min_depth > get_pos_z():
				min_depth = get_pos_z()
			move(West)
		if min_depth > get_pos_z():
			min_depth = get_pos_z()
	while (get_pos_x() !=0):
		move(West)
	while (get_pos_y() !=0):
		move(South)
	for i in range(size/2):
		#print(i) #this is just for debugging
		while (get_pos_y() != i): #goes down
			while min_depth < get_pos_z():
				dig()
			move(South)
		while (get_pos_x() != size-i): #goes right
			while min_depth < get_pos_z():
				dig()
			move(East)
		while (get_pos_y() != size-i): #goes up
			while min_depth < get_pos_z():
				dig()
			move(North)
		while (get_pos_x() != i): #goes left
			while min_depth < get_pos_z():
				dig()
			move(West)
		while min_depth < get_pos_z():
			dig()
